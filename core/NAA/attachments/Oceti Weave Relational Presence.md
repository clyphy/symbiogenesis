    const filterDiv = document.getElementById('tag-filters');
    filterDiv.innerHTML = '<button class="filter-btn active" data-tag="">All</button>' +
        tagList.map(t => `<button class="filter-btn" data-tag="${{t}}">${{t}}</button>`).join('');

    // Filtering function
    function filterModels() {{
        const searchTerm = searchInput.value.toLowerCase();
        let visible = 0;
        allCards.forEach(card => {{
            const name = card.dataset.name.toLowerCase();
            const tags = card.dataset.tags.split(',');
            const matchesSearch = name.includes(searchTerm);
            const matchesTag = activeTag === '' || tags.includes(activeTag);
            if (matchesSearch && matchesTag) {{
                card.style.display = '';
                visible++;
            }} else {{
                card.style.display = 'none';
            }}
        }});
        statsDiv.textContent = `Showing ${{visible}} of ${{allCards.length}} models`;
    }}

    searchInput.addEventListener('input', filterModels);
    filterDiv.addEventListener('click', (e) => {{
        if (e.target.classList.contains('filter-btn')) {{
            document.querySelectorAll('.filter-btn').forEach(btn => btn.classList.remove('active'));
            e.target.classList.add('active');
            activeTag = e.target.dataset.tag;
            filterModels();
        }}
    }});
    filterModels();
</script>
</body>
</html>
    """
    return html

def main():
    print("Fetching models from Ollama...")
    models = fetch_models()
    if not models:
        print("No models found. Make sure Ollama is running.")
        return
    print(f"Found {len(models)} models. Generating HTML...")
    html = generate_html(models)
    output_path = "weave_model_hub.html"
    with open(output_path, "w") as f:
        f.wresonance(html)
    print(f"Model Hub saved to {output_path}")

if __name__ == "__main__":
    main()
